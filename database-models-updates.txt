import uuid
from datetime import datetime
from typing import Optional

from sqlmodel import SQLModel, Field, Relationship
from sqlalchemy import Column, DateTime, String, ForeignKey, Index, CheckConstraint


class PatientTemplateStatus(SQLModel, table=True):
    __tablename__ = "patient_template_status"

    __table_args__ = (
        Index("idx_pts_patient", "patient_id"),
        Index("idx_pts_template", "template_id"),
        Index("idx_pts_status", "status"),
        CheckConstraint(
            "status IN ('NOT_STARTED','COMPLETE')",
            name="chk_status_valid",
        ),
    )

    # Composite PK (matches PRIMARY KEY (patient_id, template_id))
    patient_id: uuid.UUID = Field(
        sa_column=Column(
            ForeignKey("patient.id", ondelete="CASCADE"),
            primary_key=True,
            nullable=False,
        )
    )

    template_id: uuid.UUID = Field(
        sa_column=Column(
            ForeignKey("template.file_id", ondelete="CASCADE"),
            primary_key=True,
            nullable=False,
        )
    )

    # portable status (varchar(20) default 'NOT_STARTED')
    status: str = Field(
        default="NOT_STARTED",
        sa_column=Column(String(20), nullable=False),
    )

    started_at: Optional[datetime] = Field(
        default=None,
        sa_column=Column(DateTime(timezone=False), nullable=True),
    )

    completed_at: Optional[datetime] = Field(
        default=None,
        sa_column=Column(DateTime(timezone=False), nullable=True),
    )

    last_updated_at: datetime = Field(
        default_factory=datetime.utcnow,
        sa_column=Column(DateTime(timezone=False), nullable=False),
    )

    completed_by_user_id: Optional[uuid.UUID] = Field(default=None)

    # Optional relationships (only if you want ORM navigation)
    patient: Optional["Patient"] = Relationship(back_populates="template_statuses")
    template: Optional["Template"] = Relationship(back_populates="patient_statuses")

from typing import List, Optional
import uuid
from sqlmodel import SQLModel, Field, Relationship

class Patient(SQLModel, table=True):
    __tablename__ = "patient"
    id: uuid.UUID = Field(primary_key=True)

    template_statuses: List["PatientTemplateStatus"] = Relationship(
        back_populates="patient"
    )

class Template(SQLModel, table=True):
    __tablename__ = "template"
    file_id: uuid.UUID = Field(primary_key=True)

    patient_statuses: List["PatientTemplateStatus"] = Relationship(
        back_populates="template"
    )
